            feed.map((item, idx) => (
              <div key={item.id || idx} className="flex items-start gap-3 p-3 rounded-2xl bg-white/5 text-[13px]">

                      O'tib ketdi
                    </span>
                    {item.media_type && item.media_type !== 'text' && (
                      <span className="px-2 py-0.5 rounded-full text-[10px] font-bold bg-purple-500/20 text-purple-300">

                    )}

                </div>

            ))
          )}
        </div>
      </div>
      
    </div>
  );
